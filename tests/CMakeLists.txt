# 定义宏简化重复操作
function(add_test_executable test_name)
add_executable(${test_name} ${test_name}.cpp)
target_link_libraries(${test_name} PRIVATE ddla_lib)
add_test(NAME ${test_name} COMMAND ${test_name})
endfunction()
